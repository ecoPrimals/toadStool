### Pattern #2: Zero-Copy with Cow
```rust
use std::borrow::Cow;

pub struct EnvConfigLoader {
    prefix: Cow<'static, str>,  // Zero allocation for defaults!
}

impl EnvConfigLoader {
    pub fn new() -> Self {
        Self {
            prefix: Cow::Borrowed("TOADSTOOL"),  // No heap allocation
        }
    }
}
```
